<script>
let coins=0;
let highscore=0;
let score=0;
let lives=3;
let level=1;
let lane=1;
let running=false;
let gameTimer=null;
let spawnTimer=null;

let currentCar="Starter";
let currentMap="Neon City";

const unlocked={
 starter:true,
 blitz:false,
 eco:false,
 night:false,
 gold:false
};

const mapObstacles={
 city:["🚧","🛢️","🚗","🚓","🚌","🚚","🧱"],
 desert:["🪨","🌵","🛢️","🪨","🏜️"],
 snow:["❄️","🧊","🪨","🚗","☃️"],
 forest:["🌲","🪵","🪨","🦌","🌳"]
};

const pages=document.querySelectorAll(".page");

function showPage(id){
 pages.forEach(p=>p.classList.remove("active"));
 document.getElementById(id).classList.add("active");
 updateUI();
}

function updateUI(){
 document.getElementById("coins").textContent=coins;
 document.getElementById("shopCoins").textContent=coins;
 document.getElementById("statsCoins").textContent=coins;
 document.getElementById("highscore").textContent=highscore;
 document.getElementById("currentCar").textContent=currentCar;
 document.getElementById("statsCar").textContent=currentCar;
 document.getElementById("currentMap").textContent=currentMap;
 document.getElementById("statsMap").textContent=currentMap;
}

function carPosition(){
 const road=document.getElementById("gameArea");
 const player=document.getElementById("player");
 const positions=[
  road.clientWidth/6,
  road.clientWidth/2,
  road.clientWidth*5/6
 ];
 player.style.left=(positions[lane]-27)+"px";
}

function startGame(){
 stopGame();

 score=0;
 lives=3;
 level=1;
 lane=1;
 running=true;

 document.getElementById("score").textContent=score;
 document.getElementById("lives").textContent=lives;
 document.getElementById("level").textContent=level;

 carPosition();

 gameTimer=setInterval(()=>{
   if(!running)return;

   score++;
   level=1+Math.floor(score/25);

   if(score>highscore){
     highscore=score;
   }

   document.getElementById("score").textContent=score;
   document.getElementById("level").textContent=level;

   if(score===25)coins+=250;
   if(score===100)coins+=750;
   if(score===250)coins+=1500;

   updateUI();

 },1000);

 spawnTimer=setInterval(spawnObstacle,Math.max(400,900-level*35));
}

function spawnObstacle(){
 if(!running)return;

 const road=document.getElementById("gameArea");
 const obstacle=document.createElement("div");

 obstacle.className="obstacle";

 const list=mapObstacles[currentMap==="Neon City"?"city":
                            currentMap==="Desert Run"?"desert":
                            currentMap==="Snow Valley"?"snow":"forest"];

 obstacle.textContent=list[Math.floor(Math.random()*list.length)];

 const obstacleLane=Math.floor(Math.random()*3);

 const positions=[
  road.clientWidth/6,
  road.clientWidth/2,
  road.clientWidth*5/6
 ];

 obstacle.style.left=positions[obstacleLane]+"px";
 obstacle.style.top="-60px";

 road.appendChild(obstacle);

 let y=-60;

 const fall=setInterval(()=>{
   if(!running){
     clearInterval(fall);
     obstacle.remove();
     return;
   }

   y+=5+level*.55;
   obstacle.style.top=y+"px";

   const p=document.getElementById("player");
   const a=p.getBoundingClientRect();
   const b=obstacle.getBoundingClientRect();

   if(
     a.left<b.right &&
     a.right>b.left &&
     a.top<b.bottom &&
     a.bottom>b.top
   ){
     lives--;

     document.getElementById("lives").textContent=lives;

     clearInterval(fall);
     obstacle.remove();

     if(lives<=0){
       gameOver();
     }
   }

   if(y>road.clientHeight+80){
     clearInterval(fall);
     obstacle.remove();
   }

 },30);
}

function gameOver(){
 running=false;
 clearInterval(gameTimer);
 clearInterval(spawnTimer);

 coins+=Math.floor(score/10);
 updateUI();

 setTimeout(()=>{
   alert("💥 Game Over!\nPunkte: "+score+"\nCoins erhalten: "+Math.floor(score/10));
 },100);
}

function stopGame(){
 running=false;

 if(gameTimer)clearInterval(gameTimer);
 if(spawnTimer)clearInterval(spawnTimer);

 document.querySelectorAll(".obstacle").forEach(o=>o.remove());
}

document.addEventListener("keydown",e=>{
 if(!running)return;

 if(e.key==="ArrowLeft"||e.key.toLowerCase()==="a"){
   lane=Math.max(0,lane-1);
   carPosition();
 }

 if(e.key==="ArrowRight"||e.key.toLowerCase()==="d"){
   lane=Math.min(2,lane+1);
   carPosition();
 }
});

window.addEventListener("resize",carPosition);

function buyCar(car,price){
 if(unlocked[car]){
   alert("Dieses Auto ist bereits freigeschaltet.");
   return;
 }

 if(coins<price){
   alert("Du brauchst noch "+(price-coins)+" Coins.");
   return;
 }

 coins-=price;
 unlocked[car]=true;

 alert("🚗 Auto freigeschaltet!");
 updateUI();
}

function selectCar(car){
 const names={
  starter:"Starter",
  blitz:"Blitz",
  eco:"Eco Racer",
  night:"Night Rider",
  gold:"Golden GT"
 };

 if(!unlocked[car]){
   alert("Dieses Auto ist noch nicht freigeschaltet.");
   return;
 }

 currentCar=names[car];
 updateUI();
 alert(currentCar+" ausgewählt!");
}

function selectMap(map){
 const names={
  city:"Neon City",
  desert:"Desert Run",
  snow:"Snow Valley",
  forest:"Forest Road"
 };

 currentMap=names[map];
 updateUI();

 alert(currentMap+" ausgewählt!");
}

function loginAdmin(){
 const code=document.getElementById("adminCode").value;

 if(code==="260513"){
   document.getElementById("loginBox").style.display="none";
   document.getElementById("adminPanel").style.display="block";
 }else{
   alert("❌ Falscher Admin-Code");
 }
}

function addCoins(amount){
 coins+=amount;
 updateUI();
}

function addScore(amount){
 highscore+=amount;
 updateUI();
}

function unlockAll(){
 Object.keys(unlocked).forEach(k=>unlocked[k]=true);
 alert("🔓 Alle Autos freigeschaltet!");
}

function resetGame(){
 stopGame();

 coins=0;
 highscore=0;
 currentCar="Starter";
 currentMap="Neon City";

 Object.keys(unlocked).forEach(k=>{
   unlocked[k]=(k==="starter");
 });

 updateUI();

 alert("♻️ Spiel wurde zurückgesetzt.");
}

updateUI();
carPosition();
</script>
